# Experiment: Conformance Replay Receipt Validation

This experiment validates the cryptographic Replay Receipt format. A Replay Receipt is a tamper-proof certificate generated by the WebAssembly `wasm4pm` execution engine, proving that a specific event log was replayed against a specific process model with a declared fitness level.

## 1. Cryptographic Replay Receipt JSON Validation Schema

```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "CryptographicReplayReceipt",
  "type": "object",
  "properties": {
    "receipt_id": { "type": "string" },
    "timestamp": { "type": "string", "format": "date-time" },
    "execution_authority": {
      "type": "object",
      "properties": {
        "engine_identifier": { "type": "string" },
        "wasm_module_sha256": { "type": "string", "pattern": "^[0-9a-fA-F]{64}$" }
      },
      "required": ["engine_identifier", "wasm_module_sha256"]
    },
    "input_artifacts": {
      "type": "object",
      "properties": {
        "model_sha256": { "type": "string", "pattern": "^[0-9a-fA-F]{64}$" },
        "log_sha256": { "type": "string", "pattern": "^[0-9a-fA-F]{64}$" }
      },
      "required": ["model_sha256", "log_sha256"]
    },
    "replay_results": {
      "type": "object",
      "properties": {
        "fitness_score": { "type": "number", "minimum": 0.0, "maximum": 1.0 },
        "missing_tokens": { "type": "integer", "minimum": 0 },
        "remaining_tokens": { "type": "integer", "minimum": 0 },
        "produced_tokens": { "type": "integer", "minimum": 0 },
        "consumed_tokens": { "type": "integer", "minimum": 0 }
      },
      "required": ["fitness_score", "missing_tokens", "remaining_tokens", "produced_tokens", "consumed_tokens"]
    },
    "cryptographic_signature": {
      "type": "object",
      "properties": {
        "public_key": { "type": "string" },
        "signature_bytes": { "type": "string" }
      },
      "required": ["public_key", "signature_bytes"]
    }
  },
  "required": ["receipt_id", "timestamp", "execution_authority", "input_artifacts", "replay_results", "cryptographic_signature"]
}
```
